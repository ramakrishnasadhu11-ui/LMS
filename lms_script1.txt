/* =========================================================
   LMS New Machine DB Setup
   Creates:
   - LMS_Identity
   - LMS_Master
   ========================================================= */

USE [master];
GO

IF DB_ID(N'LMS_Identity') IS NULL CREATE DATABASE [LMS_Identity];
IF DB_ID(N'LMS_Master')   IS NULL CREATE DATABASE [LMS_Master];
GO

/* =========================================================
   IDENTITY DB
   ========================================================= */
USE [LMS_Identity];
GO

IF OBJECT_ID(N'[dbo].[Tenants]', N'U') IS NULL
BEGIN
    CREATE TABLE [dbo].[Tenants]
    (
        [Id]            INT IDENTITY(1,1) NOT NULL,
        [NoOfStores]    INT NOT NULL,
        [TenantId]      NVARCHAR(64) NOT NULL,
        [TenantName]    NVARCHAR(256) NOT NULL,
        [Email]         NVARCHAR(256) NOT NULL,
        [PhoneNumber]   NVARCHAR(20) NOT NULL, -- updated type
        [MiddleName]    NVARCHAR(MAX) NULL,
        [FamilyName]    NVARCHAR(MAX) NULL,
        [Address]       NVARCHAR(512) NULL,
        [City]          NVARCHAR(128) NULL,
        [Region]        NVARCHAR(128) NULL,
        [Zip]           NVARCHAR(32) NULL,
        [Country]       NVARCHAR(128) NULL,
        [CreatedDate]   DATETIME2(7) NOT NULL,
        [ModifiedDate]  DATETIME2(7) NOT NULL,
        CONSTRAINT [PK_Tenants] PRIMARY KEY CLUSTERED ([Id] ASC)
    );
END
GO

IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE name = N'IX_Tenants_Email' AND object_id = OBJECT_ID(N'[dbo].[Tenants]'))
    CREATE UNIQUE NONCLUSTERED INDEX [IX_Tenants_Email] ON [dbo].[Tenants]([Email]);
GO

IF NOT EXISTS (SELECT 1 FROM sys.indexes WHERE name = N'IX_Tenants_TenantId' AND object_id = OBJECT_ID(N'[dbo].[Tenants]'))
    CREATE UNIQUE NONCLUSTERED INDEX [IX_Tenants_TenantId] ON [dbo].[Tenants]([TenantId]);
GO

IF OBJECT_ID(N'[dbo].[TenantStoreInfos]', N'U') IS NULL

    CREATE TABLE [dbo].[TenantStoreInfos]
    (
        [Id]                INT IDENTITY(1,1) NOT NULL,
